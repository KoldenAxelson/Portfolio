{{- /* python-versions — the course's pinned stack, as a table, read straight from the requirements.txt every workbook packet ships (path in data/python/course.yaml). One file to bump; the page can't disagree with it. {{< python-versions >}} The "As of" date and the Python floor are read from that file's header comment ("As of YYYY-MM-DD", "Python 3.12+"). */ -}} {{- $source := site.Data.python.course.requirements -}} {{- $text := readFile $source -}} {{- $asOf := index (findRESubmatch `As of (\d{4}-\d{2}-\d{2})` $text 1) 0 -}} {{- $python := index (findRESubmatch `Python (\d+\.\d+)\+` $text 1) 0 -}} {{- if not (and $asOf $python) -}} {{- errorf "python-versions: %s needs \"As of YYYY-MM-DD\" and \"Python X.Y+\" in its header" $source -}} {{- end -}}
