{{- /* plate-map — which states require a front license plate, as a tile map. {{< plate-map >}} DATA is data/front-plates.yaml. Every count on this widget is derived from it here; nothing about how many states say what is typed anywhere else. ONE SQUARE PER STATE, not a geographic map, so Rhode Island's law is as easy to read as Texas's. It is complete in the HTML: colour, dashed outline and a legend carry everything. ts/plate-map.ts only adds the readout line — point at or tap a square and it names the state and what it requires, which a `title` tooltip cannot do on a phone. DROPPED states — ones that repealed the requirement — are drawn as a dashed outline, so the map shows the market getting smaller rather than only its current size. `not-prose`, because the page sets prose and Typography would restyle the list. */ -}} {{- $d := index site.Data "front-plates" -}} {{- if not $d -}} {{- errorf "plate-map: data/front-plates.yaml is missing (page: %s)" .Page.RelPermalink -}} {{- else -}} {{- $required := 0 -}} {{- $dropped := slice -}} {{- $cols := 0 -}} {{- $rows := 0 -}} {{- range $code, $s := $d.states -}} {{- if $s.required }}{{ $required = add $required 1 }}{{ end -}} {{- with $s.dropped -}} {{- $year := (time.AsTime .).Year -}} {{- $dropped = $dropped | append (dict "name" $s.name "year" $year) -}} {{- end -}} {{- $cols = math.Max $cols (add $s.col 1) -}} {{- $rows = math.Max $rows (add $s.row 1) -}} {{- end -}} {{- /* D.C. is in the data because it has a plate law; it is not a state, and "29 states" would be a number nobody can check. */ -}} {{- $requiredStates := $required -}} {{- with index $d.states "DC" }}{{ if .required }}{{ $requiredStates = sub $required 1 }}{{ end }}{{ end -}} {{- $total := len $d.states -}} {{- $dropped = sort $dropped "year" -}} {{- $headline := printf "%d states and D.C. require a front plate." $requiredStates -}} {{- end -}}
