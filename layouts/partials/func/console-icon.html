{{- /* console-icon — inline a console logo SVG from static/img/consoles/.svg, flattening its fills and strokes to currentColor so it takes the site accent color and adapts to light/dark. `none` values are preserved (so hollow shapes stay hollow), and width/height attrs are stripped so CSS controls the size (the viewBox keeps the aspect ratio). Renders nothing if the file is absent. Dict params: code (e.g. "N64"). */ -}} {{- /* Codes that share another console's logo (e.g. every Xbox generation uses the same sphere mark) alias onto that icon file instead of duplicating it. */ -}} {{- $aliases := dict "XSX" "xbox" -}} {{- $file := index $aliases .code | default (lower .code) -}} {{- $path := printf "static/img/consoles/%s.svg" $file -}} {{- if os.FileExists $path -}} {{- $s := os.ReadFile $path -}} {{- /* Park "none" outside any fill="…"/stroke="…" shape, or the recolour below eats it. */ -}} {{- $s = replace $s `fill="none"` `data-keep-fill-none` -}} {{- $s = replace $s `stroke="none"` `data-keep-stroke-none` -}} {{- $s = replaceRE `fill="[^"]*"` `fill="currentColor"` $s -}} {{- $s = replaceRE `stroke="[^"]*"` `stroke="currentColor"` $s -}} {{- $s = replace $s `data-keep-fill-none` `fill="none"` -}} {{- $s = replace $s `data-keep-stroke-none` `stroke="none"` -}} {{- $s = replaceRE `(fill|stroke):\s*#[0-9A-Fa-f]{3,8}` `$1:currentColor` $s -}} {{- $s = replaceRE ` (?:width|height)="[^"]*"` "" $s -}} {{- $s | safeHTML -}} {{- end -}}
