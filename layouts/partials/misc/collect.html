{{- /* misc/collect — every card on /misc, normalised, grouped, ordered and checked. {{ $shelves := partial "misc/collect.html" $section }} Returns content/misc/_index.md `groups:` in order, each with its `cards`. The grid (misc/list.html), the JSON-LD (schema/misc.html) and the filter (func/filter-cats.html) all read this, so they can't drift apart. Cards are the section's pages (minus `hidden: true`) plus `elsewhere:` entries (/now, /uses), and the build fails on a page with no or an unknown `group:`. Each card gets its `band` here — "live" (partials/misc/live/.html), "shot" (assets/img/misc/previews/.webp) or "icon" — and shelves sort by title, with icon-only cards last. The first shelf's first visual cards are `eager`: they are the page's largest paint. */ -}} {{- $groups := .Params.groups -}} {{- $keys := slice -}} {{- range $groups }}{{ $keys = $keys | append .key }}{{ end -}} {{- $cards := slice -}} {{- range where .Pages "Params.hidden" "!=" true -}} {{- $where := .RelPermalink -}} {{- with .File }}{{ $where = .Path }}{{ end -}} {{- if not .Params.group -}} {{- errorf "misc: %s has no `group:`. Give it one of %v (content/misc/_index.md) so /misc knows which shelf it sits on, or set `hidden: true`." $where $keys -}} {{- else if not (in $keys .Params.group) -}} {{- errorf "misc: %s sets group %q, which isn't one of %v in content/misc/_index.md." $where .Params.group $keys -}} {{- end -}} {{- $slug := path.Base (strings.TrimSuffix "/" .RelPermalink) -}} {{- $cards = $cards | append (dict "label" .Title "href" .RelPermalink "permalink" .Permalink "slug" $slug "group" .Params.group "icon" (.Params.icon | default "") "blurb" (.Params.blurb | default .Description | default "") "youtube" (.Params.youtube | default "")) -}} {{- end -}} {{- range .Params.elsewhere -}} {{- if not (in $keys .group) -}} {{- errorf "misc: elsewhere entry %q needs a `group:` from %v." .label $keys -}} {{- end -}} {{- $cards = $cards | append (dict "label" .label "href" (partial "func/url.html" .href) "permalink" (absURL .href) "slug" (path.Base (strings.TrimSuffix "/" .href)) "group" .group "icon" (.icon | default "") "blurb" (.blurb | default "") "youtube" (.youtube | default "")) -}} {{- end -}} {{- /* Live and screenshot bands rank together, so a shelf reads alphabetically whatever each card draws; only an icon-only card sinks to the end. */ -}} {{- $bandOrder := dict "live" 0 "shot" 0 "icon" 1 -}} {{- $banded := slice -}} {{- range $cards -}} {{- $band := "icon" -}} {{- if templates.Exists (printf "partials/misc/live/%s.html" .slug) }}{{ $band = "live" }} {{- else if resources.Get (printf "misc/previews/%s.webp" .slug) }}{{ $band = "shot" }}{{ end -}} {{- $banded = $banded | append (merge . (dict "band" $band "rank" (printf "%d %s" (index $bandOrder $band) .label))) -}} {{- end -}} {{- $isFirstShelf := true -}} {{- $out := slice -}} {{- range $groups -}} {{- $key := .key -}} {{- $mine := slice -}} {{- range $i, $card := sort (where $banded "group" $key) "rank" -}} {{- $isEager := and $isFirstShelf (lt $i 4) (ne $card.band "icon") -}} {{- $mine = $mine | append (merge $card (dict "eager" $isEager)) -}} {{- end -}} {{- if $mine }}{{ $isFirstShelf = false }}{{ end -}} {{- $out = $out | append (merge . (dict "cards" $mine)) -}} {{- end -}} {{- return $out -}}
