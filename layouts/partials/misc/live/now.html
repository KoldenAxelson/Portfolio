{{- /* misc/live/now — the Now card as a departure board. One flap at a time turns over to the next thing on /now: each project I'm building, then each book I'm reading, read from the page's `blocks` (the name before " — " for a project, the quoted title for a book), so the card is current whenever the page is. A live dot and the page's date sit above it. Pure HTML + CSS; the flip timing is generated for however many entries. */ -}} {{- with site.GetPage "/now" }} {{- $entries := slice -}} {{- range .Params.blocks -}} {{- $label := .label -}} {{- range .items -}} {{- if eq $label "Building" -}} {{- $entries = $entries | append (dict "label" $label "value" (index (split . " — ") 0)) -}} {{- else if eq $label "Reading" -}} {{- with findRESubmatch `"([^"]+)"` . 1 }}{{ $entries = $entries | append (dict "label" $label "value" (index (index . 0) 1)) }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- $n := len $entries -}} {{- $hold := 3 -}} {{- $cycle := mul $n $hold -}} {{- $show := div 100.0 $n -}} {{- $flip := div (mul 100.0 0.45) $cycle -}} Now{{ with .Params.updated }} · {{ (time.AsTime .).Format "Jan 2006" }}{{ end }} {{- range $i, $e := $entries }} {{ $e.label }} {{ $e.value }} {{- end }} {{- end }}
