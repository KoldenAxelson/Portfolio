{{- /* misc/live/writing — the Writing card as a storybook: each story's own illustration, one after another with a slow push in, captioned with the story's title and its collection. Stories are found in the collections under content/misc/writing/: every "## Title" followed by a writing/illustration shortcode is one slide, so a new story with a picture joins the card. Pure HTML + CSS; the crossfade timing is generated for however many slides there are. Without animation (reduced motion) the first slide stays. */ -}} {{- $slides := slice -}} {{- with site.GetPage "/misc/writing" -}} {{- range .RegularPages.ByWeight -}} {{- $collection := .Title -}} {{- range findRESubmatch `(?m)^## (.+)\n+\{\{< writing/illustration "([^"]+)"` .RawContent -1 -}} {{- $title := index . 1 -}} {{- with resources.Get (printf "writing/%s.webp" (index . 2)) -}} {{- $slides = $slides | append (dict "title" $title "collection" $collection "img" .) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- with $slides }} {{- $n := len . -}} {{- $hold := 4 -}}{{- /* seconds per slide */ -}} {{- $cycle := mul $n $hold -}} {{- $show := div 100.0 $n -}} {{- $fade := div (mul 100.0 0.8) $cycle -}}{{- /* a 0.8s crossfade, as a share of the cycle */ -}} {{- range $i, $s := . }} {{- $one := $s.img.Fill "480x300 webp q78" }}{{ $two := $s.img.Fill "960x600 webp q72" }} {{ $s.title }} {{ $s.collection }} {{- end }} {{- end }}
