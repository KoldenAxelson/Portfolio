{{- /* skyrim-verify — assert the properties the potion builder depends on, at build time. Called once from skyrim-assets.html. These two invariants were checked by hand when the 183-ingredient table was compiled, then not again. A hand-edit that broke either would have shipped silently and shown up as a builder that quietly omits an effect or produces mixtures the game cannot make. `errorf` fails the build instead. 1. Every ingredient has exactly four DISTINCT effects. Alchemy has no notion of a fifth, and a duplicate would let one ingredient satisfy the two-carriers rule on its own. 2. Every effect is carried by at least two ingredients. An effect on only one could never reach a bottle, so offering it in the picker is a dead end. Also checks that every effect an ingredient names actually exists in effects.yaml, which is what keeps the payload's numeric indices meaningful. */ -}} {{- $known := dict -}} {{- range site.Data.skyrim.effects }}{{ $known = merge $known (dict .name true) }}{{ end -}} {{- $carriers := dict -}} {{- range $slug, $ing := site.Data.skyrim.ingredients -}} {{- $effects := $ing.effects | default slice -}} {{- if ne (len $effects) 4 -}} {{- errorf "skyrim data: %q has %d effects, expected exactly 4" $slug (len $effects) -}} {{- end -}} {{- if ne (len $effects) (len (uniq $effects)) -}} {{- errorf "skyrim data: %q lists the same effect twice: %v" $slug $effects -}} {{- end -}} {{- range $effects -}} {{- if not (index $known .) -}} {{- errorf "skyrim data: %q lists effect %q, which is not in effects.yaml" $slug . -}} {{- end -}} {{- $carriers = merge $carriers (dict . (add (int (index $carriers . | default 0)) 1)) -}} {{- end -}} {{- end -}} {{- /* 3. An ingredient may only carry a MULTIPLIER for an effect it actually has. builder.ts leans on this when it drops a mixture whose ingredients can be reduced: if a passenger could deviate on an effect it does not carry, removing it could change a magnitude, and the search would be filtering out real answers. */ -}} {{- range $effect := site.Data.skyrim.effects -}} {{- range $slug, $_ := $effect.mult -}} {{- $ing := index site.Data.skyrim.ingredients $slug -}} {{- if not $ing -}} {{- errorf "skyrim data: effect %q has a multiplier for %q, which is not an ingredient" $effect.name $slug -}} {{- else if not (in ($ing.effects | default slice) $effect.name) -}} {{- errorf "skyrim data: %q has a multiplier for %q but does not carry it" $slug $effect.name -}} {{- end -}} {{- end -}} {{- end -}} {{- range site.Data.skyrim.effects -}} {{- $count := int (index $carriers .name | default 0) -}} {{- if lt $count 2 -}} {{- errorf "skyrim data: effect %q is carried by %d ingredient(s); it needs 2 to ever appear in a potion" .name $count -}} {{- end -}} {{- end -}} {{- /* 4. THE BUILD AGREES WITH THE TREES. Two things a hand-maintained build drifts on, both silent without this: a. every perk id John spends on exists in the tree it is filed under. A renamed id would just stop being drawn — a taken perk quietly rendering as untaken. b. `take:` equals the ranks those perks actually cost. The budget bars, the total under them and the page's own "42 perks" all read from `take:`, so a build that claims 9 and spends 8 puts a wrong number in three places at once. Also asserts the ids are unique per tree, since a duplicate would inflate the sum here and be invisible in the drawing. */ -}} {{- with site.Data.skyrim.john -}} {{- range $row := .build -}} {{- $tree := index site.Data.skyrim.perks $row.skill -}} {{- if not $tree -}} {{- errorf "skyrim data: john.yaml builds %q, which is not a tree in perks.yaml" $row.skill -}} {{- else -}} {{- $ranks := dict -}} {{- range $tree.perks -}} {{- if index $ranks .id -}} {{- errorf "skyrim data: perks.yaml lists %q twice in the %s tree" .id $row.skill -}} {{- end -}} {{- $ranks = merge $ranks (dict .id (.ranks | default 1)) -}} {{- end -}} {{- $spent := 0 -}} {{- range $id := ($row.perks | default slice) -}} {{- $cost := index $ranks $id -}} {{- if not $cost -}} {{- errorf "skyrim data: john.yaml takes %q in %s, which is not a perk in that tree" $id $row.skill -}} {{- else -}} {{- $spent = add $spent $cost -}} {{- end -}} {{- end -}} {{- if ne $spent $row.take -}} {{- errorf "skyrim data: john.yaml claims take: %d for %s but its perks cost %d" $row.take $row.skill $spent -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
